"""Команда для запуска в фоне:"""
docker compose up --build -d

"""Проверка статуса: Чтобы увидеть, какие контейнеры запущены и всё ли с ними в порядке"""
docker compose ps

"""Просмотр логов: Поскольку консоль теперь свободна, логи не выводятся в реальном времени.
Если что-то пошло не так (например, FastAPI не стартует из-за ошибок в main.py), посмотрите логи командой:
Флаг -f позволит следить за ними в реальном времени, выйти можно через Ctrl+C"""
docker compose logs -f

"""Остановка проекта: Чтобы остановить и удалить контейнеры, запущенные в фоне:"""
docker compose down
